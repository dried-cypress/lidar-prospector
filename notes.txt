Prospector training stack: ingest a small Historic England area, build LiDAR-backed training examples, train a local archaeology model, and use that model during normal Prospector analysis.

docker compose up -d
docker compose build --no-cache app trainer
docker compose up -d app trainer
docker compose run --rm trainer prospector train init
docker compose run --rm trainer prospector train ingest-he --xmin 518000 --ymin 104000 --xmax 528000 --ymax 114000
docker compose run --rm trainer prospector train build-dataset --download-lidar --xmin 518000 --ymin 104000 --xmax 528000 --ymax 114000 --tile-size 1000 --max-tiles 25 --max-features 100
docker compose run --rm trainer prospector train fit --dataset project/training/datasets/<dataset>/training-dataset.npz
docker compose run --rm trainer prospector train status
prospector analyse --latitude 50.8620 --longitude -0.2547
